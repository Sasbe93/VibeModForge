cmake_minimum_required(VERSION 3.20)
project(isomtool CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CD ${CMAKE_CURRENT_SOURCE_DIR}/external/Chkdraft/src)
include_directories(${CD} ${CD}/mapping_core ${CMAKE_CURRENT_SOURCE_DIR}/external/RareCpp/include /usr/local/include)
file(GLOB MC ${CD}/mapping_core/*.cpp)
list(FILTER MC EXCLUDE REGEX "lite_")
add_executable(isomtool main.cpp ${MC} ${CD}/cross_cut/logger.cpp ${CD}/cross_cut/simple_icu.cpp)
target_compile_definitions(isomtool PRIVATE NOMINMAX)
target_compile_options(isomtool PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/gle.h)
target_link_libraries(isomtool /usr/local/lib/libstorm.a /usr/local/lib/libcasc.a bz2 z pthread icuuc)
